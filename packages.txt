# Packages installed by install.sh. This file is the source of truth for both
# the installation and the drift check (./install.sh --check-drift).
#
# Format: <manager> <name> [tags...]   (`#` starts a comment)
#   manager : brew | cask | go | npm | krew | command
#   name    : package name as the manager lists it after installation
#             (Homebrew renames like kubectl -> kubernetes-cli must use the new name,
#              tap-qualified formulae are compared by their last path element)
#   tags    : os=<macos|ubuntu>  only on the OS
#             personal           only in personal mode (-p)
#             skip-ci            skip in CI because the installation takes long
#             src=<module>       module path for `go install` (required for go)
#
# `command` entries are installed by OS specific functions in os-*/setup.sh,
# so they are only checked for their presence.
